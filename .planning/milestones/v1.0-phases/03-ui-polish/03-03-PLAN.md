
    **Part B: Create a11y audit spec** at `tests/e2e/a11y.spec.ts`:
    This Playwright spec covers every page route with axe-core scans:
    ```tsx
    import { test, expect } from "@playwright/test";
    import AxeBuilder from "@axe-core/playwright";

    test.describe("Accessibility audit", () => {
      const PAGES = [
        { path: "/", name: "Home" },
        { path: "/changes/new", name: "New benchmark change" },
        { path: "/benchmarks", name: "Benchmark catalog" },
        { path: "/benchmark-aanvraag", name: "New benchmark request" },
        { path: "/admin/client-config", name: "Client config" },
        { path: "/updates", name: "Updates" },
      ];

      for (const page of PAGES) {
        test(`${page.name} has no critical or serious violations`, async ({ page: p }) => {
          await p.goto(page.path);
          await p.waitForLoadState("networkidle");

          const results = await new AxeBuilder({ page: p }).analyze();

          expect(results.violations.filter(
            (v) => v.impact === "critical" || v.impact === "serious"
          )).toEqual([]);
        });
      }
    });
    ```

    **Part C: Run the audit**
    ```bash
    npx playwright test tests/e2e/a11y.spec.ts --reporter=list
    ```

    **Part D: Fix findings** — Based on the UI-SPEC contrast audit, pre-emptively fix known issues:

    In globals.css, change accent color for body-size text on white backgrounds per UI-SPEC § Color → Accent Contrast Remediation:
    - `.button-ghost` `color: var(--accent)` → `color: var(--accent-deep)` (body-size text on white button bg)

    - `.catalog-section a` (inline style `color: var(--accent)`) → change to `color: var(--accent-deep)`
    - Any other `color: var(--accent)` applied to ≤18px text on white background → switch to `--accent-deep`
    
    If the Playwright scan reveals violations beyond these pre-emptive fixes, fix them in the same task. Typical fixes: color contrast, missing form labels, missing heading hierarchy.
