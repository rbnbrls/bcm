    Replace the inline navigation in layout.tsx with a dedicated NavBar client component.

    **1. `components/navbar.tsx`** (NEW FILE)
    
    `"use client"` directive — required for `usePathname()` hook. Per Next.js App Router rules, only client components can use client hooks. The NavBar is the ONLY client component in the layout hierarchy.
    
    ```typescript
    "use client";
    
    import Link from "next/link";
    import { usePathname } from "next/navigation";
    
    const NAV_ITEMS = [
      { label: "Dashboard", href: "/" },
      { label: "Wijzigingen", href: "/changes" },
      { label: "Rapportages", href: "/reports" },
      { label: "Beheer", href: "/admin" },
    ] as const;
    
    export function NavBar() {
      const pathname = usePathname();
    
      return (
        <nav aria-label="Hoofdnavigatie">
          {NAV_ITEMS.map(({ label, href }) => {
            // Dashboard (/) must match exactly — not prefix-match (would match ALL paths)
            // Other items match when pathname starts with their href
            const isActive = href === "/"
              ? pathname === "/"
              : pathname.startsWith(href);
    
            return (
              <Link
                key={href}
                href={href}
                aria-current={isActive ? "page" : undefined}
                className={isActive ? "nav-link--active" : undefined}
              >
                {label}
              </Link>
            );
          })}
        </nav>
      );
    }
    ```

    Key design decisions (per D-XX, UI-SPEC Interaction Contract):
    - **4 nav items** — Dashboard, Wijzigingen, Rapportages, Beheer (removed: Nieuwe change, Benchmark catalogus, Wijzigingshistorie — now accessible via dashboard category cards)
    - **"Dashboard" active only on exact /** — avoids highlighting dashboard on all pages
    - **Prefix matching for others** — `/changes` matches `/changes`, `/changes/new`, `/changes/123`, etc. Same for `/reports` and `/admin`
    - **aria-current="page"** — accessibility attribute on active link per WCAG best practices
    - **nav-link--active class** — consumed by CSS from Plan 1's globals.css additions (`.topbar nav a.nav-link--active { color: var(--accent-deep); font-weight: 700; }`)
    - **`as const`** — type-safe readonly tuple for NAV_ITEMS
    - **No `"use client"` in layout.tsx** — the NavBar is the only client component, preventing the entire layout from becoming client-rendered
